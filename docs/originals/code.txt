```markdown
# MODULE 1 — Introduction to One Health and Pandemic Forecasting
### Concept
**One Health** recognizes that human, animal, and environmental health are interconnected — most emerging infectious diseases (Ebola, avian influenza, COVID-19) originate at that interface.

**Surveillance** is the continuous collection and analysis of health data to detect abnormal patterns. An **outbreak** is a localized increase in cases above what is expected; an **epidemic** is a wider increase across a region; a **pandemic** is a global epidemic. **Forecasting** estimates future case counts from historical patterns; **early warning** flags when observed activity deviates enough from the expected baseline to warrant investigation.
```

```python
# =============================================================================
# MODULE 5 — Data Cleaning and Preprocessing
# =============================================================================

# 1. Handle negative case counts (retrospective corrections)
# Clip negatives to 0. Do not drop them automatically; investigate first.
df["new_cases"] = df["new_cases"].clip(lower=0)

# 2. Resample to Weekly Totals
# This smooths day-of-week reporting artifacts (e.g., weekend under-reporting)
# We create a copy to work with a clean index
weekly = (
    df.set_index("date")[["new_cases"]]
    .resample("W")          # 'W' stands for Week (Sun-Sat)
    .sum()
    .reset_index()
    .rename(columns={"new_cases": "weekly_cases"})
)

# 3. Interpolate short internal gaps (do NOT interpolate large gaps)
# limit=2 fills gaps of max 2 weeks. 
# Larger gaps will remain NaN and be flagged.
weekly["weekly_cases"] = weekly["weekly_cases"].interpolate(
    method='linear', limit=2, limit_direction="both"
)

# 4. Flag remaining missing weeks rather than silently filling them
weekly["is_imputed_gap"] = weekly["weekly_cases"].isna()

print("Weekly Resampled Data Head:")
print(weekly.head())
print("\nMissing values remaining:")
print(weekly.isnull().sum())
```

```python
# =============================================================================
# MODULE 7 — Time-Series Analysis Fundamentals
# =============================================================================

# Prepare series (drop NaNs for decomposition)
series = weekly.set_index("date")["weekly_cases"].dropna()

# 1. Decomposition (Trend + Seasonal + Residual)
# Period=52 assumes yearly seasonality in weekly data
decomposition = seasonal_decompose(series, model='additive', period=52)
decomposition.plot()
plt.show()

# 2. Stationarity Test (Augmented Dickey-Fuller)
adf_result = adfuller(series)
print(f"ADF Statistic: {adf_result[0]:.3f}")
print(f"P-value: {adf_result[1]:.3f}")

if adf_result[1] < 0.05:
    print("Result: Reject non-stationarity (Series is likely stationary)")
else:
    print("Result: Series is likely non-stationary (has trend/unit root)")

# 3. Autocorrelation (ACF / PACF)
fig, axes = plt.subplots(1, 2, figsize=(14, 4))
plot_acf(series, lags=52, ax=axes[0])
plot_pacf(series, lags=20, ax=axes[1])
plt.show()
```

```python
# =============================================================================
# MODULE 10 — Feature Engineering
# =============================================================================

fe = weekly.copy()

# 1. Lag Features (Past 1, 2, 3, 4 weeks)
for lag in [1, 2, 3, 4]:
    fe[f"lag_{lag}"] = fe["weekly_cases"].shift(lag)

# 2. Rolling Statistics (Mean, Std over 4 weeks)
# CRITICAL: Shift(1) is used to avoid data leakage (using current week to predict current week)
fe["rolling_mean_4"] = fe["weekly_cases"].shift(1).rolling(4).mean()
fe["rolling_std_4"] = fe["weekly_cases"].shift(1).rolling(4).std()

# 3. Calendar Features
fe["week_of_year"] = fe["date"].dt.isocalendar().week.astype(int)
fe["month"] = fe["date"].dt.month

# Drop rows containing NaN generated by lags/rollings
fe = fe.dropna().reset_index(drop=True)

print("Feature Engineered Dataset:")
print(fe.head())
```

```python
# =============================================================================
```
```python
# MODULE 12 — MODEL TRAINING (Part B: SARIMA)
# =============================================================================

# SARIMA works on the raw series index, not the feature matrix
# Prepare series
sarima_series = weekly.set_index("date")["weekly_cases"]

# Define Model
# Order(p,d,q)=(1,1,1), Seasonal Order(P,D,Q,S)=(1,1,1,52)
# enforce_stationarity/invertibility=False handles potential math issues in short data
sarima_model = SARIMAX(
    sarima_series,
    order=(1, 1, 1),
    seasonal_order=(1, 1, 1, 52),
    enforce_stationarity=False,
    enforce_invertibility=False
)

# Fit
sarima_fit = sarima_model.fit(disp=False) # disp=False suppresses verbose output

# Forecast length of test set
forecast_steps = len(test)
sarima_forecast_obj = sarima_fit.get_forecast(steps=forecast_steps)
sarima_preds = sarima_forecast_obj.predicted_mean
sarima_ci = sarima_forecast_obj.conf_int()
```

```python
# =============================================================================
# MODULE 13 — Model Evaluation
# =============================================================================

def evaluate(y_true, y_pred, label):
    mae = mean_absolute_error(y_true, y_pred)
    rmse = np.sqrt(mean_squared_error(y_true, y_pred))
    
    # Symmetric MAPE (handles zero/near-zero cases better)
    smape = 100 * np.mean(2 * np.abs(y_pred - y_true) / (np.abs(y_true) + np.abs(y_pred)))
    
    print(f"{label:15s} | MAE={mae:.1f} | RMSE={rmse:.1f} | sMAPE={smape:.2f}%")
    return {"label": label, "MAE": mae, "RMSE": rmse, "sMAPE": smape}

# Reset indices to ensure alignment for evaluation
y_test_reset = y_test.reset_index(drop=True)
rf_preds_s = pd.Series(rf_preds).reset_index(drop=True)
sarima_preds_s = sarima_preds.reset_index(drop=True)

# Evaluate All Models
evaluate(y_test_reset, baseline_df.loc[y_test.index, "moving_avg"], "Moving Avg Baseline")
evaluate(y_test_reset, rf_preds_s, "Random Forest")
evaluate(y_test_reset, sarima_preds_s, "SARIMA")
```

```python
# =============================================================================
# MODULE 15 — Outbreak Detection and Early Warning
# =============================================================================

# Use Training data to establish baseline statistics
# Rolling 8-week window to calculate expected range
baseline_mean = train["weekly_cases"].rolling(8).mean()
baseline_std = train["weekly_cases"].rolling(8).std()

# Align baseline to full series for calculation
# (Reindex to match weekly index)
weekly_analysis = weekly.copy()
weekly_analysis["baseline_mean"] = baseline_mean.reindex(weekly_analysis.index)
weekly_analysis["baseline_std"] = baseline_std.reindex(weekly_analysis.index)

# Fill forward any NaNs introduced by rolling start
weekly_analysis[["baseline_mean", "baseline_std"]] = weekly_analysis[["baseline_mean", "baseline_std"]].fillna(method="ffill")

# Calculate Z-Score: How many standard deviations is current value from baseline?
weekly_analysis["z_score"] = (
    (weekly_analysis["weekly_cases"] - weekly_analysis["baseline_mean"]) 
    / weekly_analysis["baseline_std"]
)
```
